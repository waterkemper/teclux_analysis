+
+    public const STATUS_PENDING = 'pending';
+
+    public const STATUS_COMPLETED = 'completed';

+        string $idempotencyKey,
+        string $payloadHash,
+        ?int $operadorCodigo = null,
+        ?string $requestId = null,
+    ): array {

+        if ($existing !== null) {
+            $status = (string) $existing->status;
+            $sameHash = (string) $existing->payload_hash === $payloadHash;
+
+            if ($status === self::STATUS_COMPLETED) {
+                if (! $sameHash) {
+                    throw ValidationException::withMessages([
+                        'idempotency_key' => 'Esta chave de idempotência já foi usada com outro payload.',
+                    ]);
+                }
+
+                /** @var array<string, mixed> $decoded */
+                $decoded = json_decode((string) ($existing->result_json ?? '{}'), true) ?: [];
+                $decoded['idempotent'] = true;
+
+                return ['kind' => 'replay', 'result' => $decoded];
+            }
+
+            $this->delete($origem, $acao, $idempotencyKey);
+        }
+
+        try {
+            DB::table(self::TABLE)->insert([
+                'origem' => $origem,
+                'acao' => $acao,
+                'idempotency_key' => $idempotencyKey,
+                'payload_hash' => $payloadHash,
+                'status' => self::STATUS_PENDING,

+                'request_id' => $requestId,
+                'created_at' => now(),
+                'updated_at' => now(),
+            ]);
+        } catch (QueryException) {
+            $again = $this->find($origem, $acao, $idempotencyKey);
+            if ($again !== null

+                && (string) $again->status === self::STATUS_COMPLETED
+            ) {
+                /** @var array<string, mixed> $decoded */
+                $decoded = json_decode((string) ($again->result_json ?? '{}'), true) ?: [];
+                $decoded['idempotent'] = true;

+                return ['kind' => 'replay', 'result' => $decoded];
+            }

+                ]);
+            }
+
+            throw ValidationException::withMessages([

+    }

+            ->where('acao', $acao)

+<?php
+
+declare(strict_types=1);

+namespace App\Application\Vendas\Contratos\Cadastro;
+
+use Illuminate\Support\Facades\DB;
+
+/**
+ * OCC do cabeçalho `contratos`: PostgreSQL `xmin`; SQLite `cloud_row_version`.
+ */
+final class ContratoCadastroRevisionGuard
+{
+    public function findRevisionToken(string $numero): ?string
+    {
+        $numero = $this->padNumero($numero);
+
+        if (DB::connection()->getDriverName() === 'pgsql') {
+            $one = DB::selectOne('SELECT xmin::text AS v FROM contratos WHERE numero = ?', [$numero]);
+
+            return isset($one->v) ? (string) $one->v : null;
+        }
+
+        $v = DB::table('contratos')->where('numero', $numero)->value('cloud_row_version');
+
+        return $v !== null ? (string) $v : null;
+    }
+
+    public function lockWithExpectedRevision(string $numero, string $expectedRevision): bool
+    {
+        $numero = $this->padNumero($numero);
+        $expected = trim($expectedRevision);
+        if ($expected === '') {
+            return false;
+        }
+
+        if (DB::connection()->getDriverName() === 'pgsql') {
+            $one = DB::selectOne(
+                'SELECT xmin::text AS v FROM contratos WHERE numero = ? AND xmin::text = ? FOR UPDATE',
+                [$numero, $expected],
+            );
+
+            return $one !== null;
+        }
+
+        $current = DB::table('contratos')->where('numero', $numero)->lockForUpdate()->first();
+        if ($current === null) {
+            return false;
+        }
+
+        $currentVersion = isset($current->cloud_row_version) ? (int) $current->cloud_row_version : 0;
+
+        return (string) $currentVersion === $expected;
+    }
+
+    public function padNumero(string $numero): string
+    {
+        $trimmed = trim($numero);
+        if ($trimmed === '') {
+            return $numero;
+        }
+
+        return str_pad($trimmed, 20, ' ', STR_PAD_RIGHT);
+    }

+declare(strict_types=1);

+namespace App\Domain\Vendas\Contratos;

+/**
+ * Reconhecimento de arestas da máquina documental (Etapa 1: sem efeitos).
+ */
+final class ContratoTransitionGraph
+{
+    /**
+     * @return list<array{from: string, to: string, recognized: bool, implemented: bool}>
+     */
+    public function edgesFrom(?ContratoSituacaoDocumental $from): array
+    {
+        if ($from === null) {
+            return [];
+        }
+
+        $candidates = match ($from) {
+            ContratoSituacaoDocumental::Orcado => ['R', 'C', 'F'],
+            ContratoSituacaoDocumental::Reservado => ['F', 'C', 'O'],
+            ContratoSituacaoDocumental::Faturado => ['R', 'C'],
+            ContratoSituacaoDocumental::Cancelado => ['O', 'R', 'F'],
+            ContratoSituacaoDocumental::NotaParcial,
+            ContratoSituacaoDocumental::NotaFiscal => ['C', 'O', 'R', 'F'],
+        };
+
+        $out = [];
+        foreach ($candidates as $to) {
+            $recognized = $this->isRecognized($from->value, $to);
+            $out[] = [
+                'from' => $from->value,
+                'to' => $to,
+                'recognized' => $recognized,
+                'implemented' => false,
+            ];
+        }
+
+        return $out;
+    }
+
+    public function isRecognized(string $from, string $to): bool
+    {
+        $from = strtoupper(trim($from));
+        $to = strtoupper(trim($to));
+
+        return match (true) {
+            $from === 'O' && $to === 'R' => true,
+            $from === 'R' && $to === 'F' => true,
+            $from === 'O' && $to === 'C' => true,
+            $from === 'R' && $to === 'C' => true,
+            $from === 'F' && $to === 'C' => true,
+            $from === 'F' && $to === 'R' => true,
+            default => false,
+        };
+    }
+
+    public function isImplemented(string $from, string $to): bool
+    {
+        unset($from, $to);
+
+        // Etapa 1: nenhuma transição de domínio além de create→O (fora deste grafo).
+        return false;
+    }

+import { useCallback, useRef, useState } from 'react';
+import axios from 'axios';
+import type { SectionId } from '@/types/contratoCadastro';
+
+type CacheEntry = {
+  revision: string;
+  data: unknown;
+};
+
+/**
+ * Cache client-side (numero + revision + section). Invalida em conflito 409 / nova revision.
+ */
+export function useContratoSectionCache(numero: string | null, revision: string | null) {
+  const cacheRef = useRef<Map<string, CacheEntry>>(new Map());
+  const [loadingSection, setLoadingSection] = useState<SectionId | null>(null);
+  const [sectionError, setSectionError] = useState<string | null>(null);
+
+  const cacheKey = useCallback(
+    (section: SectionId) => `${numero ?? 'draft'}|${revision ?? ''}|${section}`,
+    [numero, revision],
+  );
+
+  const invalidate = useCallback(() => {
+    cacheRef.current.clear();
+  }, []);
+
+  const loadSection = useCallback(
+    async (section: SectionId, url: string): Promise<unknown | null> => {
+      if (!numero || !revision) {
+        return null;
+      }
+      const key = cacheKey(section);
+      const hit = cacheRef.current.get(key);
+      if (hit && hit.revision === revision) {
+        return hit.data;
+      }
+
+      setLoadingSection(section);
+      setSectionError(null);
+      try {
+        const res = await axios.get(url, { params: { expected_revision: revision } });
+        const data = res.data?.data ?? res.data;
+        cacheRef.current.set(key, { revision, data });
+        return data;
+      } catch (err: unknown) {
+        const status = axios.isAxiosError(err) ? err.response?.status : undefined;
+        if (status === 409) {
+          invalidate();
+          setSectionError('O Contrato foi alterado. Recarregue antes de continuar.');
+        } else {
+          setSectionError('Não foi possível carregar esta seção.');
+        }
+        return null;
+      } finally {
+        setLoadingSection(null);
+      }
+    },
+    [cacheKey, invalidate, numero, revision],
+  );
+
+  return { loadSection, invalidate, loadingSection, sectionError, setSectionError };
